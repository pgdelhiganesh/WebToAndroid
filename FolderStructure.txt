Ex: Usage in javascript file

function saveExcelFromBase64(base64, filename) {
    const encoded = encodeURIComponent(base64);
    const fname = encodeURIComponent(filename);
    const intentUrl = `intent://saveExcelFromBase64?filename=${fname}&data=${encoded}#Intent;scheme=webtoandroid;package=com.opsym.webtoandroid;end`;
    window.location.href = intentUrl;
}
